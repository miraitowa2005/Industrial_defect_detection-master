### 错误原因

YOLO 试图去这里找验证集图片：
`D:\MachineLearning\ComputerVersion\Industrial_defect_detection-master\dataset\NEU-DET\validation\images`
**但是电脑告诉它：这个文件夹不存在。**

这通常是因为你的数据集文件夹名字叫 `test` 或者 `val`，而不是我之前代码里假设的 `validation`。

请打开你的文件资源管理器，去 `D:\MachineLearning\ComputerVersion\Industrial_defect_detection-master\dataset\NEU-DET\` 目录下看一眼。

#### 方法二：一键重组数据集（推荐！最稳妥）

工业数据集往往结构很乱（有的图片在 `train/crazing/` 子文件夹里，有的在 `IMAGES` 里）。YOLO 要求结构非常严格（`images/train` 和 `labels/train` 平级）。

为了**彻底解决**路径找不到、图片找不到的问题，我写了一个**终极整理脚本**。它会把你的数据自动整理成 YOLO 标准格式，并自动生成一个新的 yaml 文件。

**请在项目根目录下新建 `auto_organize_yolo.py`，粘贴并在 conda 环境中运行：**

```python
import os
import shutil
import random
import glob
import yaml
from pathlib import Path

# === 配置区域 ===
# 你的原始数据根目录 (请确保这个路径是对的)
SOURCE_ROOT = r"D:\MachineLearning\ComputerVersion\Industrial_defect_detection-master\dataset\NEU-DET"
# 我们将把整理好的数据放到这个新目录，避免弄乱原数据
TARGET_ROOT = r"D:\MachineLearning\ComputerVersion\Industrial_defect_detection-master\yolo_dataset_final"

def main():
    # 1. 创建 YOLO 标准目录结构
    for split in ['train', 'val']:
        os.makedirs(os.path.join(TARGET_ROOT, 'images', split), exist_ok=True)
        os.makedirs(os.path.join(TARGET_ROOT, 'labels', split), exist_ok=True)
    
    print(f"🚀 正在扫描原始数据: {SOURCE_ROOT} ...")
    
    # 2. 暴力搜索所有 jpg 图片 (不管它在多少层子文件夹里)
    # 这一步是为了解决你的图片可能散落在 crazing/, patches/ 等文件夹的问题
    all_images = glob.glob(os.path.join(SOURCE_ROOT, '**', '*.jpg'), recursive=True)
    all_images = [p for p in all_images if 'yolo_dataset_final' not in p] # 排除掉目标文件夹
    
    print(f"✅ 找到了 {len(all_images)} 张图片")
    if len(all_images) == 0:
        print("❌ 错误：没找到任何 .jpg 图片，请检查 SOURCE_ROOT 路径是否正确！")
        return

    # 3. 随机划分 训练集(80%) / 验证集(20%)
    random.shuffle(all_images)
    split_idx = int(len(all_images) * 0.8)
    train_imgs = all_images[:split_idx]
    val_imgs = all_images[split_idx:]
    
    print(f"📊 划分计划：训练集 {len(train_imgs)} 张，验证集 {len(val_imgs)} 张")

    # 4. 开始复制文件
    def process_files(img_list, split_name):
        count = 0
        for img_path in img_list:
            # 基础文件名 (e.g., crazing_1.jpg)
            basename = os.path.basename(img_path)
            stem = os.path.splitext(basename)[0] # crazing_1
            
            # 对应的 txt 标签路径 (假设在之前的 labels 文件夹里，或者同级目录)
            # 我们尝试在 dataset 下递归寻找同名的 .txt
            # 这里简化逻辑：去原来的 'labels' 文件夹找，或者假设就在同级
            
            # 策略：在原始根目录下找同名txt
            # 注意：之前 xml2yolo.py 生成的 txt 可能在 labels 文件夹
            possible_txt = list(Path(SOURCE_ROOT).rglob(f"{stem}.txt"))
            
            src_txt = None
            if len(possible_txt) > 0:
                src_txt = possible_txt[0] # 找到对应的txt
            
            if src_txt and src_txt.exists():
                # 复制图片
                shutil.copy2(img_path, os.path.join(TARGET_ROOT, 'images', split_name, basename))
                # 复制标签
                shutil.copy2(src_txt, os.path.join(TARGET_ROOT, 'labels', split_name, f"{stem}.txt"))
                count += 1
            else:
                # 如果没标签，这张图就废了，跳过
                pass
        return count

    print("Copying train files...")
    train_count = process_files(train_imgs, 'train')
    print("Copying val files...")
    val_count = process_files(val_imgs, 'val')
    
    print(f"✅ 整理完成！实际有效(有图有标签)数据：训练集 {train_count}, 验证集 {val_count}")

    # 5. 自动生成 yaml 文件
    yaml_content = {
        'path': TARGET_ROOT.replace('\\', '/'), # 绝对路径，防止出错
        'train': 'images/train',
        'val': 'images/val',
        'nc': 6,
        'names': ['crazing', 'inclusion', 'patches', 'pitted_surface', 'rolled-in_scale', 'scratches']
    }
    
    yaml_path = os.path.join(os.getcwd(), 'my_dataset.yaml')
    with open(yaml_path, 'w') as f:
        yaml.dump(yaml_content, f, sort_keys=False)
    
    print(f"🎉 配置文件已生成：{yaml_path}")
    print("👉 请修改 train_yolo.py，将 data='neu_det.yaml' 改为 data='my_dataset.yaml'")

if __name__ == '__main__':
    main()

```

### 执行步骤

1. 运行上面的脚本：`python auto_organize_yolo.py`。
* 它会把你乱七八糟的文件整理到一个新文件夹 `yolo_dataset_final`。
* 它会在根目录生成一个新的 `my_dataset.yaml`。


2. 修改你的 `train_yolo.py`：
```python
# 修改 data 参数
results = model.train(
    data='my_dataset.yaml',  # <--- 改成新的配置文件
    # ... 其他不变
)

```


3. 再次运行 `train_yolo.py`。这次一定能跑起来！
